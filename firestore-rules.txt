rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {

    function signedIn() { return request.auth != null; }
    function teamDoc(t) { return get(/databases/$(database)/documents/teams/$(t)); }
    function isMember(t) { return signedIn() && exists(/databases/$(database)/documents/teams/$(t)/members/$(request.auth.uid)); }
    function isOwner(t) { return signedIn() && teamDoc(t).data.owner == request.auth.uid; }

    // 내 팀 목록 (나만 읽고 씀)
    match /users/{uid} {
      allow read, write: if signedIn() && request.auth.uid == uid;
    }

    // 초대 코드: 코드를 아는 사람만 하나씩 열어 볼 수 있음 (목록 보기 불가)
    match /invites/{code} {
      allow get: if signedIn();
      allow create: if isOwner(request.resource.data.teamId);
      allow delete: if isOwner(resource.data.teamId);
    }

    match /teams/{t} {
      allow create: if signedIn() && request.resource.data.owner == request.auth.uid;
      allow read: if isMember(t);
      // 코치는 팀 이름·설정만 바꿀 수 있고, 감독 정보·잠금 열쇠·초대 코드는 감독만
      allow update: if isMember(t)
        && request.resource.data.owner == resource.data.owner
        && (isOwner(t) || !request.resource.data.diff(resource.data).affectedKeys().hasAny(['owner', 'crypto', 'recovery', 'check', 'inviteCode', 'level']));
      allow delete: if isOwner(t);

      match /members/{uid} {
        allow read: if isMember(t);
        allow create: if signedIn() && request.auth.uid == uid && (
          (teamDoc(t).data.owner == uid && request.resource.data.role == 'owner')
          || (request.resource.data.role == 'coach'
              && request.resource.data.code is string
              && get(/databases/$(database)/documents/invites/$(request.resource.data.code)).data.teamId == t));
        allow update: if isOwner(t);
        allow delete: if signedIn() && (request.auth.uid == uid || isOwner(t));
      }

      // 학부모 공지판: 로그인 없이 읽기
      match /notices/{id} {
        allow read: if true;
        allow write: if isMember(t);
      }
      match /public/{id} {
        allow read: if true;
        allow write: if isMember(t);
      }

      // 선수·경기·일정·일지·학업·장비 (학생 정보는 암호문으로만 저장됨)
      match /{col}/{id} {
        allow read, write: if col != 'members' && isMember(t);
      }
    }
  }
}
